Change 1 — Font size applies only to selected text
The font size buttons currently set fontSize state which changes the entire editor. Instead:
If text is selected: wrap only the selected text in <span style="font-size: Xpx"> using execCommand("fontSize", "7") then replace the resulting <font size="7"> elements with <span style="font-size: Xpx"> elements

The key is to restore the saved selection from savedRangeRef.current before calling execCommand, since e.preventDefault() on mousedown may have shifted focus.
Replace the font size button onMouseDown with:
tsx
onMouseDown={(e) => {
  e.preventDefault();
  const saved = savedRangeRef.current;
  const div = contentRef.current;
  if (saved && !saved.collapsed && div) {
    div.focus();
    const sel = window.getSelection();
    if (sel) { sel.removeAllRanges(); sel.addRange(saved); }
    document.execCommand("fontSize", false, "7");
    div.querySelectorAll("font[size='7']").forEach((el) => {
      const span = document.createElement("span");
      span.style.fontSize = `${size}px`;
      while (el.firstChild) span.appendChild(el.firstChild);
      el.replaceWith(span);
    });
    scheduleAutosave();
    updateWordCount();
    localStorage.setItem("folio_font_size", size.toString());
  } else {
    setFontSize(size);
    localStorage.setItem("folio_font_size", size.toString());
  }
}}

onMouseDown={(e) => {
  e.preventDefault();

  if (saved && !saved.collapsed && div && div.contains(saved.commonAncestorContainer)) {
